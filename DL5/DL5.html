epochs = 20
batch_size = 64
noise_dim = 100

for epoch in range(epochs):
    print(f"Epoch {epoch+1}/{epochs}")

    for i in range(train_x.shape[0] // batch_size):

        # Train Discriminator
        noise = np.random.normal(0, 1, (batch_size, noise_dim))
        fake_images = generator.predict(noise, verbose=0)

        real_images = train_x[i*batch_size:(i+1)*batch_size]

        labels_real = np.ones((batch_size, 1))
        labels_fake = np.zeros((batch_size, 1))

        discriminator.trainable = True
        d_loss_real = discriminator.train_on_batch(real_images, labels_real)
        d_loss_fake = discriminator.train_on_batch(fake_images, labels_fake)

        # Train Generator
        noise = np.random.normal(0, 1, (batch_size, noise_dim))
        labels_gen = np.ones((batch_size, 1))

        discriminator.trainable = False
        g_loss = gan.train_on_batch(noise, labels_gen)

    # Show output every 10 epochs
    if epoch % 10 == 0:
        noise = np.random.normal(0, 1, (10, noise_dim))
        gen_images = generator.predict(noise, verbose=0)

        plt.figure(figsize=(10,4))
        for k in range(10):
            plt.subplot(2,5,k+1)
            plt.imshow(gen_images[k].reshape(28,28), cmap='gray')
            plt.axis('off')
        plt.show()
